  export function selectBusinessDescription(b: any, locale: string): string | null {
    if (!b) return null
    if (locale !== 'en') {
      const v = b[`description_${locale}`]

### 3.3 번역본 필드를 코드에 추가/연동할 최적의 설계 방안 (Next Steps)

마스터 DB를 건드리지 않고 성능 및 유지보수성을 극대화하기 위한 연동 지점:

1. **업체 소개문 연동**:
   - **위치**: `lib/i18n/dictionaries.ts:522`의 `selectBusinessDescription()`
   - **방식**: `data/i18n/descriptions/{locale}.json` (slug 키 기반의 맵)을 생성하고, `selectBusinessDescription` 내부에서 먼저 해당 맵을 조회하도록 오버레이.
   ```typescript
   // lib/i18n/dictionaries.ts (개념 예시)
   import descZh from '@/data/i18n/descriptions/zh.json'
   import descRu from '@/data/i18n/descriptions/ru.json'
   import descJa from '@/data/i18n/descriptions/ja.json'
   import descHi from '@/data/i18n/descriptions/hi.json'

   const I18N_DESCRIPTIONS: Record<string, Record<string, string>> = {
     zh: descZh, ru: descRu, ja: descJa, hi: descHi
   }

   export function selectBusinessDescription(b: any, locale: string): string | null {
     if (!b) return null
     // 1. 다국어 오버레이 사전 우선 조회
     if (I18N_DESCRIPTIONS[locale]?.[b.slug]) {
       return I18N_DESCRIPTIONS[locale][b.slug]
     }
     // 2. 기존 필드 조회 (ko, en 등)
     if (locale !== 'en') {
       const v = b[`description_${locale}`]
       if (hasText(v)) return String(v).trim()
     }
     // 3. 비한국어 로케일에서는 한국어로 유출되지 않도록 description_ko 폴백 차단
     if (['zh', 'ru', 'ja', 'hi'].includes(locale)) {
       return b.description_en || null
     }
     return b.description_en || b.description_vi || b.description_ko || null
   }
   ```
2. **메뉴 명칭 연동**:
   - **타입 확장 (`lib/menus.ts:31-48`)**:
     `MenuItem`에 `name_zh?: string; name_ru?: string; name_ja?: string; name_hi?: string;` 추가.
   - **변환 매핑 (`lib/menus.ts:130-170`)**:
     `menu_translations.json` (원문 키 또는 place_id+원문 키)을 로드하여 `MenuItem` 생성 시 주입.
   - **컴포넌트 렌더링 (`components/BusinessMenuSection.tsx:300-325`)**:
     ```typescript
     const localizedName = (item as any)[`name_${locale}`]
     const primaryName = localizedName || (isKo ? (item.name_ko || item.name_original) : (item.name_en || item.name_original))
     ```
   - **한국어 누락 방지 가드**: 비한국어 로케일에서는 절대 `item.name_ko`를 폴백하지 않고 `item.name_en || item.name_original`을 보장.
